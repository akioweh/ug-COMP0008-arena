Consider the following code.

```java

03:     String id;
04:     Reader(ReadWriteController controller, String name) {

13:                 ctrl.releaseRead();
14:             }
15:         } catch (InterruptedException e){}
16:     }
17: }
18: public class Writer extends Thread {
19:     ReadWriteController ctrl;
20:     String id;
21:     Writer(ReadWriteController controller, String name) {
22:         ctrl = controller;
23:         id = name;
24:     }
25:     public void run() {
26:         try {
27:             while(true) {
28:                 ctrl.acquireWrite();
29:                 Thread.sleep(1000);
30:                 ctrl.releaseWrite();
31:             }
32:         } catch (InterruptedException e){}
33:     }
34: }
35: public class ReadWriteController {
36:     private int readers = 0;
37:     private boolean writing = false;
38:     private int numWaitingWriters = 0;
39:     private boolean readersTurn = false;
40:     synchronized public void acquireRead()
41:                             throws InterruptedException {
42:         while (writing || (numWaitingWriters>0 && !readersTurn)) {
43:             wait();
44:         }
45:         ++readers;
46:     }
47:     synchronized public void releaseRead() {
48:         --readers;
49:         readersTurn=false;

52:     synchronized public void acquireWrite()
53:                             throws InterruptedException {
54:         ++numWaitingWriters;
55:         while (readersTurn || writing) { wait(); }
56:         --numWaitingWriters;
57:         writing = true;
58:     }
59:     synchronized public void releaseWrite() {
60:         writing = false;
61:         readersTurn = true;
62:         notifyAll();
63:     }
64: }
```
